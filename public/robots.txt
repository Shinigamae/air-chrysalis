# The archive is deliberately not discoverable yet — the same decision as the
# `noindex, nofollow` meta tag in src/layouts/AppShell.astro, said again here.
#
# This file only does anything on a host that serves the site from the root of its
# own hostname, which is Azure Static Web Apps. On GitHub Pages this site lives under
# /air-chrysalis/, and a crawler only ever reads robots.txt from the origin root — so
# there the meta tag is the whole story and this is inert.
#
# To make the site public: delete this file, drop the robots meta tag in AppShell, and
# re-add the sitemap integration in astro.config.mjs.
#
# Link-preview crawlers are let through, and only them: they fetch a page when
# someone pastes its link, read the og: tags, and build the card. LinkedInBot in
# particular obeys this file, so without these lines a link shared on LinkedIn
# shows up as a bare URL. None of them is a search engine, and the noindex
# header and meta tag still apply to anything that is.
User-agent: LinkedInBot
User-agent: facebookexternalhit
User-agent: Twitterbot
User-agent: Slackbot-LinkExpanding
User-agent: Discordbot
User-agent: TelegramBot
User-agent: WhatsApp
Allow: /

User-agent: *
Disallow: /
